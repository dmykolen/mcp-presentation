````md magic-move {at:3, lines: true}
async def run(mcp_servers: list[MCPServer], directory_path: str):
    agent = Agent(
        name="Assistant",
        instructions=f"Answer questions about the git repository at {directory_path}, use that for repo_path",
        mcp_servers=mcp_servers,
    )

    while True:
        message = input("Enter your message (or type 'exit' to quit): ")
        if message.lower() == 'exit':
            print("Exiting...")
            break

        print("\n" + "-" * 40)
        print(f"Running: {message}")
        result = await Runner.run(starting_agent=agent, input=message)
        print(result.final_output)
```

async def main():
    directory_path = input("Please enter the path to the git repository: ")

    async with MCPServerStdio(
        cache_tools_list=True,  # Cache the tools list for efficiency
        params={"command": "uvx", "args": ["mcp-server-git"]},
    ) as server1, MCPServerStdio(
        cache_tools_list=True,
        params={"command": "npx", "args": ["-y", "@modelcontextprotocol/server-filesystem", "/app/python-proj/"]},
    ) as server2:
        with trace(workflow_name="MCP Git Example"):
            await run([server1, server2], directory_path)


if __name__ == "__main__":
    if not shutil.which("uvx"):
        raise RuntimeError("uvx is not installed. Please install it with `pip install uvx`.")

    asyncio.run(main())
````
